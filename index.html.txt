<script>
    function showStep(n) {
        document.querySelectorAll('[id^="step"]').forEach(s => s.style.display = 'none');
        document.getElementById('step' + n).style.display = 'block';
        if(n === 13) { 
            document.getElementById('noBtn').style.display = 'none'; 
        }
    }

    function moveButton() {
        const btn = document.getElementById('noBtn');
        const maxX = window.innerWidth - btn.clientWidth - 30;
        const maxY = window.innerHeight - btn.clientHeight - 30;
        
        btn.style.left = Math.floor(Math.random() * maxX) + 'px';
        btn.style.top = Math.floor(Math.random() * maxY) + 'px';
        btn.style.transform = 'none'; // عشان يتحرك بحرية تامة
    }
</script>
